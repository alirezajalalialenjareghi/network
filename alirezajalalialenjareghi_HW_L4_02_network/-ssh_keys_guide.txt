################################################################################
# راهنمای تولید و استفاده از کلید SSH (RSA 2048-bit)
################################################################################

# این فایل شامل مراحل تولید کلید SSH، فرمت‌دهی کلید عمومی،
# توضیح نحوه احراز هویت بدون رمز عبور (passwordless authentication)
# و نمونه‌ای از فایل پیکربندی ~/.ssh/config برای تسهیل اتصالات است.

################################################################################
# 1. تولید کلید SSH (RSA 2048-bit)
################################################################################

# برای تولید یک جفت کلید SSH (خصوصی و عمومی) با الگوریتم RSA و طول 2048 بیت،
# از دستور زیر در ترمینال خود استفاده کنید:

ssh-keygen -t rsa -b 2048

# توضیح دستور:
# -   ssh-keygen: ابزار خط فرمان برای ایجاد، مدیریت و تبدیل کلیدهای اح خراز هویت SSH.
# -   -t rsa: نوع کلید را مشخص می‌کند (در اینجا RSA). الگوریتم‌های دیگر مانند ed25519 نیز قابل استفاده هستند.
# -   -b 2048: طول کلید را بر حسب بیت مشخص می‌کند (2048 بیت برای RSA توصیه می‌شود).

# پس از اجرای دستور، از شما دو سوال پرسیده می‌شود:
# 1. Enter file in which to save the key (/home/user/.ssh/id_rsa):
#    - مسیر پیش‌فرض را بپذیرید (با فشردن Enter) یا مسیر دلخواه خود را وارد کنید.
#      معمولاً کلید خصوصی در `~/.ssh/id_rsa` و کلید عمومی در `~/.ssh/id_rsa.pub` ذخیره می‌شوند.
# 2. Enter passphrase (empty for no passphrase):
#    - این بخش اختیاری است. وارد کردن یک passphrase (رمز عبور) امنیتی اضافی ایجاد می‌کند.
#      هر بار که از کلید خصوصی استفاده می‌کنید، این passphrase از شما خواسته می‌شود.
#      برای احراز هویت بدون رمز عبور (passwordless authentication)، این فیلد را خالی بگذارید (فقط Enter را بزنید).
#    - اگر passphrase وارد کنید، امنیت کلید خصوصی شما در صورت لو رفتن فایل، بیشتر حفظ می‌شود.

# خروجی پس از اجرای موفقیت‌آمیز:
# Generating public/private rsa key pair.
# Enter file in which to save the key (/home/user/.ssh/id_rsa):
# Enter passphrase (empty for no passphrase):
# Enter same passphrase again:
# Your identification has been saved in /home/user/.ssh/id_rsa.
# Your public key has been saved in /home/user/.ssh/id_rsa.pub.
# The key fingerprint is:
# SHA256:xxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx user@hostname
# The key's randomart image is:
# +---[RSA 2048]----+
# |        ...      |
# |       . .       |
# |      . . .      |
# |     . + . .     |
# |    . = S . .    |
# |   . * B o .     |
# |  . + = * O .    |
# |   . . o B .     |
# |    . . . .      |
# +----[SHA256]-----+

# پس از اجرای این دستور، دو فایل در مسیر مشخص شده (معمولاً `~/.ssh/`) ایجاد می‌شود:
# - `id_rsa`: کلید خصوصی شما (بسیار محرمانه است، هرگز آن را به اشتراک نگذارید).
# - `id_rsa.pub`: کلید عمومی شما (می‌توانید آن را با دیگران به اشتراک بگذارید).

################################################################################
# 2. Public key را به فرمت مناسب در بیاورید
################################################################################

# کلید عمومی (Public Key) معمولاً به صورت یک خط متنی طولانی ذخیره می‌شود
# که شامل نوع کلید، محتوای رمزنگاری شده کلید و در نهایت نام کاربری@نام هاست (comment) است.
# برای مشاهده و کپی کردن کلید عمومی، از دستور زیر استفاده کنید:

cat ~/.ssh/id_rsa.pub

# خروجی نمونه (این خط را کپی کنید):
# ssh-rsa AAAAB3NzaC1yc2EAAAADAQABAAACAQC...<بخش طولانی رمزنگاری شده>... user@hostname

# فرمت استاندارد کلید عمومی SSH به شکل زیر است:
# <key_type> <base64-encoded-key-data> [comment]
# مثال: ssh-rsa AAAAB3NzaC1yc2E... user@hostname

# این فرمت برای اضافه کردن به فایل `authorized_keys` در سرور راه دور مناسب است.
# اگر نیاز به کپی کردن آن برای استفاده در سرویس‌های ابری (مانند GitHub, GitLab, AWS) دارید،
# همین فرمت یک خطی کافی است.

################################################################################
# 3. توضیح دهید چگونه این key برای passwordless authentication استفاده میشود
################################################################################

# احراز هویت بدون رمز عبور (Passwordless Authentication) با استفاده از جفت کلید SSH به روش زیر کار می‌کند:

# 1.  **تولید جفت کلید:** شما یک جفت کلید (خصوصی و عمومی) تولید می‌کنید.
#     -   کلید خصوصی (`id_rsa`) روی سیستم **شما** باقی می‌ماند و باید کاملاً محرمانه نگه داشته شود.
#     -   کلید عمومی (`id_rsa.pub`) به سرور **راه دور** منتقل می‌شود.

# 2.  **انتقال کلید عمومی به سرور:** کلید عمومی خود را به سرور راه دور اضافه می‌کنید.
#     -   معمولاً با استفاده از دستور `ssh-copy-id`:
#         ssh-copy-id -i ~/.ssh/id_rsa.pub user@remote_server_ip
#     -   این دستور کلید عمومی شما را به انتهای فایل `~/.ssh/authorized_keys` در حساب کاربری `user` روی `remote_server_ip` اضافه می‌کند.
#     -   اگر `ssh-copy-id` در دسترس نیست، می‌توانید کلید عمومی را دستی کپی و در فایل `~/.ssh/authorized_keys` در سرور راه دور جای‌گذاری کنید.
#         (مطمئن شوید که مجوزهای فایل `~/.ssh` روی 700 و فایل `~/.ssh/authorized_keys` روی 600 باشد).

# 3.  **فرآیند اتصال:** وقتی شما تلاش می‌کنید از طریق SSH به سرور راه دور متصل شوید:
#     -   کلاینت SSH شما (روی سیستم محلی) ابتدا کلید خصوصی (`id_rsa`) را پیدا می‌کند.
#     -   سرور راه دور یک "چالش" (challenge) تصادفی ایجاد کرده و آن را به کلاینت شما ارسال می‌کند.
#     -   کلاینت SSH شما با استفاده از کلید خصوصی، این چالش را رمزنگاری (امضا) کرده و پاسخ رمزنگاری شده را به سرور برمی‌گرداند.
#     -   سرور راه دور، کلید عمومی شما را که قبلاً در `authorized_keys` ذخیره شده است، برمی‌دارد.
#     -   سرور با استفاده از کلید عمومی خود، پاسخ دریافتی از کلاینت شما را رمزگشایی می‌کند.
#     -   اگر رمزگشایی موفقیت‌آمیز باشد (یعنی پاسخ دریافتی با آنچه سرور انتظار دارد مطابقت داشته باشد)، احراز هویت تایید می‌شود و شما بدون نیاز به وارد کردن رمز عبور سرور، وارد می‌شوید.

# **نکته مهم:** اگر هنگام تولید کلید، passphrase تعریف کرده باشید، SSH از شما آن passphrase را خواهد پرسید (نه رمز عبور سرور). برای حذف کامل نیاز به وارد کردن passphrase، می‌توانید از ابزارهایی مانند `ssh-agent` استفاده کنید.

################################################################################
# 4. فایل ~/.ssh/config برای راحتی connection بنویسید
################################################################################

# فایل `~/.ssh/config` به شما امکان می‌دهد تنظیمات اتصال SSH را برای هاست‌های مختلف
# به صورت خلاصه و قابل مدیریت تعریف کنید. این کار باعث می‌شود دستورات SSH شما کوتاه‌تر و خواناتر شوند.

# برای ایجاد یا ویرایش این فایل، از یک ویرایشگر متن استفاده کنید:
# nano ~/.ssh/config
# یا
# vim ~/.ssh/config

# نمونه محتوای فایل `~/.ssh/config`:

# -------------------- START OF ~/.ssh/config --------------------
# Host shortcuts for easier SSH connections

# --- سرور اصلی ---
Host myserver
    HostName 192.168.1.100       # آدرس IP یا نام دامنه سرور
    User myuser                 # نام کاربری شما در سرور راه دور
    Port 22                     # پورت SSH (پیش‌فرض 22 است، اگر متفاوت است بنویسید)
    IdentityFile ~/.ssh/id_rsa  # مسیر کلید خصوصی مورد استفاده (اگر پیش‌فرض نیست)
    # ForwardAgent yes          # برای فعال کردن agent forwarding (اختیاری)
    # LogLevel INFO             # سطح جزئیات لاگ (اختیاری)

# --- سرور دیگر ---
Host another-server
    HostName vps.example.com
    User admin
    Port 2222                   # پورت SSH متفاوت
    IdentityFile ~/.ssh/keys/another_server_key # مسیر کلید خصوصی متفاوت

# --- سرور با دسترسی از طریق Bastion Host ---
# این بخش نمونه‌ای از اتصال زنجیره‌ای (ProxyJump) است.
# ابتدا به bastion وصل شده، سپس از آنجا به internal-server می‌رویم.
Host bastion
    HostName bastion.example.com
    User admin_bastion
    IdentityFile ~/.ssh/bastion_key

Host internal-server
    HostName 10.0.0.5             # آدرس IP داخلی
    User appuser
    ProxyJump bastion           # از طریق هاست 'bastion' متصل شو
    # IdentityFile ~/.ssh/internal_key # اگر کلید متفاوتی نیاز است

# --------------------- END OF ~/.ssh/config ---------------------

# توضیح پارامترهای رایج:
# Host: یک نام مستعار دلخواه برای اتصال. با دستور `ssh <نام مستعار>` استفاده می‌شود.
# HostName: آدرس IP یا نام دامنه واقعی سرور.
# User: نام کاربری پیش‌فرض برای اتصال به سرور.
# Port: شماره پورت SSH سرور (اگر با پورت پیش‌فرض 22 متفاوت است).
# IdentityFile: مسیر فایل کلید خصوصی که باید برای احراز هویت استفاده شود.
# ForwardAgent yes: فعال کردن Agent Forwarding. این امکان را می‌دهد که کلید SSH سیستم محلی شما در سرور راه دور نیز قابل استفاده باشد (مثلاً برای اتصال به سرور سوم).
# ProxyJump <host>: برای اتصال به سرور مقصد از طریق یک یا چند سرور واسط (Jump Host).

# پس از ذخیره این فایل، می‌توانید به سادگی با دستور زیر به سرور خود متصل شوید:
# ssh myserver
# یا
# ssh another-server
# یا
# ssh internal-server
